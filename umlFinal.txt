Classe Fond {
  - idFond : String
  - typeFond : Enum {Banque, Caisse}
  - solde : Decimal
  - devise : Monnaie

  + crediter(montant : Decimal)
  + debiter(montant : Decimal)
  + getSolde() : Decimal
}

  + genererReleve()
